cmake_minimum_required(VERSION 3.25)
project(dq8-native VERSION 0.0.1 LANGUAGES C CXX
        DESCRIPTION "Native PC port of Dragon Quest VIII (PS2) by static recompilation")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

include(Dq8Options)
include(CompilerWarnings)
include(Sanitizers)

if(DQ8_BUILD_TESTS)
    enable_testing()
endif()

add_subdirectory(third_party)
if(DQ8_BUILD_RECOMP)
    include(Ps2Recomp)
endif()
add_subdirectory(src/core)
add_subdirectory(src/runtime)
add_subdirectory(src/game)
add_subdirectory(src/host)
add_subdirectory(src/cli)
if(DQ8_BUILD_GS_PARALLEL)
    add_subdirectory(tools/gs)
endif()
if(DQ8_BUILD_TESTS)
    add_subdirectory(tests/unit)
    add_subdirectory(tests/local)
    add_subdirectory(tests/abi)
endif()
if(DQ8_BUILD_E2E)
    add_subdirectory(tests/e2e)
endif()
